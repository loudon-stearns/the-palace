{\anton\fontsize{26}{28}\selectfont FIREFLIES FALLING INTO STEP}\par\vspace{6pt}
{\large Each firefly keeps its own pace, and leans toward the crowd's.}\par\vspace{10pt}
\[
  \frac{d\theta_i}{dt} \;=\; \omega_i \;+\; \frac{K}{N}\sum_{j=1}^{N} \sin\!\left(\theta_j-\theta_i\right)
  \qquad\qquad
  r\,e^{i\psi} \;=\; \frac{1}{N}\sum_{j=1}^{N} e^{i\theta_j}
\]
\vspace{4pt}
{\manrope\small\color{ink3} $\theta_i$ the phase of firefly $i$ \quad $\omega_i$ its own pace \quad $K$ how hard they pull \quad $r$ how in step}\par\vspace{8pt}
{\ttfamily\footnotesize\color{red} N = 7 \quad K = 2 \quad r: 0.19 → 0.99 in 12 s}\par{\ttfamily\scriptsize\color{ink3} from the simulation in Shop/FLUX (Hugging Face)/proofs/2026-11-15-split-way}
